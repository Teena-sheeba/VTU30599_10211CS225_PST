
import java.util.*;

class Product {
    int id;
    String name;

    Product(int id, String name) {
        this.id = id;
        this.name = name;
    }
}

class Order implements Comparable<Order> {
    int orderId;
    int priority;

    Order(int orderId, int priority) {
        this.orderId = orderId;
        this.priority = priority;
    }

    public int compareTo(Order o) {
        return o.priority - this.priority;
    }
}

public class Main {
    public static void main(String[] args) {
        ArrayList<Product> products = new ArrayList<>();
        HashMap<Integer, Product> productMap = new HashMap<>();
        PriorityQueue<Order> orders = new PriorityQueue<>();

        products.add(new Product(101, "Laptop"));
        products.add(new Product(102, "Mobile"));
        products.add(new Product(103, "Headphones"));

        for (Product p : products) {
            productMap.put(p.id, p);
        }

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter product ID to search: ");
        int id = sc.nextInt();

        if (productMap.containsKey(id)) {
            System.out.println("Product: " +
                    productMap.get(id).name);
        } else {
            System.out.println("Product not found");
        }

        orders.add(new Order(1, 2));
        orders.add(new Order(2, 5));
        orders.add(new Order(3, 1));

        System.out.println("Orders by priority:");

        while (!orders.isEmpty()) {
            Order o = orders.poll();
            System.out.println("Order " + o.orderId +
                    " Priority " + o.priority);
        }

        sc.close();
    }
}
