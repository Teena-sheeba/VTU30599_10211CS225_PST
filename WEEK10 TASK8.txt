
import java.util.HashSet;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        HashSet<String> student1 = new HashSet<>();
        HashSet<String> student2 = new HashSet<>();

        System.out.print("Enter number of subjects for Student 1: ");
        int n1 = sc.nextInt();
        sc.nextLine();

        System.out.println("Enter Student 1 subjects:");
        for (int i = 0; i < n1; i++) {
            student1.add(sc.nextLine());
        }

        System.out.print("Enter number of subjects for Student 2: ");
        int n2 = sc.nextInt();
        sc.nextLine();

        System.out.println("Enter Student 2 subjects:");
        for (int i = 0; i < n2; i++) {
            student2.add(sc.nextLine());
        }

        student1.retainAll(student2);

        System.out.println("Common subjects: " + student1);

        sc.close();
    }
}


SAMPLE INPUT
Enter number of subjects for Student 1: 3
Enter Student 1 subjects:
Java
Python
DBMS
Enter number of subjects for Student 2: 3
Enter Student 2 subjects:
Python
DBMS
Networks


SAMPLE OUTPUT
Common subjects: [Python, DBMS]
